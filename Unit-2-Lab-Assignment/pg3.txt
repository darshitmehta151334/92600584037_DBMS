CREATE OR REPLACE PROCEDURE DELETE_EMP(
EMP_ID IN EMP1.EID%TYPE
)AS

BEGIN
    DELETE FROM EMP1
    WHERE EID=EMP_ID;

EXCEPTION
    WHEN OTHERS THEN
        DBMS_OUTPUT.PUT_LINE(TO_CHAR(EMP_ID) ||' NO DATA FOND ');
END;
/
