SET SERVEROUTPUT ON;

DECLARE
    celsius    NUMBER;
    fahrenheit NUMBER;
BEGIN
    -- Accept input
    celsius := &celsius;

    -- Convert Celsius to Fahrenheit
    fahrenheit := (celsius * 9 / 5) + 32;

    -- Display result
    DBMS_OUTPUT.PUT_LINE('Celsius    = ' || celsius);
    DBMS_OUTPUT.PUT_LINE('Fahrenheit = ' || fahrenheit);
END;
/
