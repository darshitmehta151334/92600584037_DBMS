6. Write a PL/SQL block which accepts measurement in feet and displays it in cm, inch and
meter.
--------------------------------------------------------------------------------------------------------------------------

SET SERVEROUTPUT ON;
DECLARE
feet REAL;
inch REAL;
cm REAL;
meter REAL;
BEGIN
-- Accept input
feet := &feet;
-- Conversion
inch := feet * 12;
cm := feet * 30.48;
meter := feet * 0.3048;
-- Display results
DBMS_OUTPUT.PUT_LINE('Measurement in Feet : ' || feet);
DBMS_OUTPUT.PUT_LINE('Measurement in Inches : ' || inch);
DBMS_OUTPUT.PUT_LINE('Measurement in CM : ' || cm);
DBMS_OUTPUT.PUT_LINE('Measurement in Meters : ' || meter);
END;
