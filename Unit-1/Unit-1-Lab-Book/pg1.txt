1. Write a PL/SQL block that find:
1. Area of Circle
2. Area of Rectangle
3. Circumference of Circle
------------------------------------------------------
SET SERVEROUTPUT ON;
DECLARE
r NUMBER;
l NUMBER;
b NUMBER;
area_circle NUMBER;
area_rectangle NUMBER;
circumference NUMBER;
BEGIN
r := &radius;
l := &length;
b := &breadth;
area_circle := 3.14 * r * r;
area_rectangle := l * b;
circumference := 2 * 3.14 * r;
DBMS_OUTPUT.PUT_LINE('Area of Circle = ' || area_circle);
DBMS_OUTPUT.PUT_LINE('Area of Rectangle = ' || area_rectangle);
DBMS_OUTPUT.PUT_LINE('Circumference of Circle = ' || circumference);
END;
/SET SERVEROUTPUT OFF
SET FEEDBACK ON
SET VERIFY ON
