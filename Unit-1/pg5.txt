CREATE TABLE EMP(
 EID NUMBER(3) PRIMARY KEY,
 EName VARCHAR(15),
 Deptno NUMBER(3),
 Deptname VARCHAR(30),
 Gender VARCHAR(5),
 Age NUMBER(3),
 BasicSal NUMBER(10)
);

SET SERVEROUTPUT ON;

DECLARE
    id EMP.EID%TYPE;
    name EMP.EName%TYPE;
    basic EMP.BasicSal %TYPE;
	
    hra        NUMBER;
    da         NUMBER;
    medical NUMBER := 500;
    pf          NUMBER;
    gross     NUMBER;

BEGIN

 -- Take Employee ID from user
    id := &Enter_id;
    
 -- Fetch employee details

    SELECT EName, Basicsal
    INTO name, basic
    FROM EMP
    WHERE EID = id;

 -- Calculate salary components
    hra := basic * 0.15;
    da  := basic * 0.50;
    pf   := basic * 0.10;

 -- Calculate Gross Salary
    gross := basic + da + hra + medical - pf;

-- Display Result --
    DBMS_OUTPUT.PUT_LINE('--------------------------------------');
    DBMS_OUTPUT.PUT_LINE('Employee ID     : ' || id);
    DBMS_OUTPUT.PUT_LINE('Employee Name   : ' || name);
    DBMS_OUTPUT.PUT_LINE('Basic Salary    : ' || basic);
    DBMS_OUTPUT.PUT_LINE('DA              : ' || da);
    DBMS_OUTPUT.PUT_LINE('HRA             : ' || hra);
    DBMS_OUTPUT.PUT_LINE('Medical         : ' || medical);
    DBMS_OUTPUT.PUT_LINE('PF              : ' || pf);
    DBMS_OUTPUT.PUT_LINE('Gross Salary    : ' || gross);
    DBMS_OUTPUT.PUT_LINE('--------------------------------------');

EXCEPTION
    WHEN NO_DATA_FOUND THEN
        DBMS_OUTPUT.PUT_LINE('Employee ID not found.');
END;
