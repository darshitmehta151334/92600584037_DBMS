4.1 Program to print numbers 1 to 100 using FOR loop

//DECLARE 
      // i NUMBER;
BEGIN
   FOR i IN 1..100 LOOP
      DBMS_OUTPUT.PUT_LINE(i);
   END LOOP;
END;
/

4.2 Program to print numbers 1 to 100 using LOOP command

DECLARE
   i NUMBER := 1;
BEGIN
   LOOP
      DBMS_OUTPUT.PUT_LINE(i);
      i := i + 1;
      EXIT WHEN i > 100;
   END LOOP;
END;
/

4.3 Program to print numbers 1 to 100 using WHILE loop

DECLARE
   i NUMBER := 1;
BEGIN
   WHILE i <= 100 LOOP
      DBMS_OUTPUT.PUT_LINE(i);
      i := i + 1;
   END LOOP;
END;
/
