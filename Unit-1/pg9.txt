SET SERVEROUTPUT ON;

DECLARE
    p  NUMBER;
    r  NUMBER;
    n  NUMBER;
    si NUMBER;
BEGIN
    -- Accept inputs
    p := &principal;
    r := &rate;
    n := &years;

    -- Calculate Simple Interest
    si := (p * r * n) / 100;

    -- Display result
    DBMS_OUTPUT.PUT_LINE('Simple Interest = ' || si);
END;
/
