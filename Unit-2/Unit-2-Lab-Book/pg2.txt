SET SERVEROUTPUT ON;

CREATE OR REPLACE PROCEDURE SEARCH_EMP(
    P_EMP_ID   IN EMP1.EID%TYPE,
    P_EMP_NAME OUT EMP1.ENAME%TYPE,
    P_EMP_SAL  OUT EMP1.BASICSAL%TYPE,
    P_EMP_DNO  OUT EMP1.DEPTNO%TYPE
)
AS
BEGIN

    SELECT ENAME, BASICSAL, DEPTNO
    INTO P_EMP_NAME, P_EMP_SAL, P_EMP_DNO
    FROM EMP1
    WHERE EID = P_EMP_ID;

EXCEPTION
    WHEN NO_DATA_FOUND THEN
        DBMS_OUTPUT.PUT_LINE('EMPLOYEE ID ' || P_EMP_ID || ' NOT FOUND');

END;
/

DECLARE
    G_NAME EMP1.ENAME%TYPE;
    G_SAL  EMP1.BASICSAL%TYPE;
    G_DNO  EMP1.DEPTNO%TYPE;
BEGIN

    SEARCH_EMP(102, G_NAME, G_SAL, G_DNO);
        DBMS_OUTPUT.PUT_LINE('EMPLOYEE NAME: ' || G_NAME);
        DBMS_OUTPUT.PUT_LINE('BASIC SALARY: ' || G_SAL);
        DBMS_OUTPUT.PUT_LINE('DEPARTMENT NO: ' || G_DNO);
END;
/
