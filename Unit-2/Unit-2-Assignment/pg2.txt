7. Create a function that returns the annual salary of an employee
(salary × 12). Call the function inside a SELECT statement.
-----------------------------------------------------------------------------------------

CREATE TABLE EMPLOYEE (
EMP_ID NUMBER,
EMP_NAME VARCHAR2(50),
SALARY NUMBER
);
SELECT EMP_ID,
EMP_NAME,
SALARY,
ANNUAL_SALARY(SALARY) AS ANNUAL_SALARY
FROM EMPLOYEE;

CREATE OR REPLACE FUNCTION ANNUAL_SALARY(P_SALARY NUMBER)
RETURN NUMBER
IS
BEGIN
RETURN P_SALARY * 12;
END;
/
SELECT ANNUAL_SALARY(25000) AS ANNUAL_SALARY
FROM DUAL;
