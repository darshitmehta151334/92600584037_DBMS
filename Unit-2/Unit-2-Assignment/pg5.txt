10. Create a function that returns employee experience category based
on joining date.
Categories:
 Less than 2 Years → Fresher
 2–5 Years → Experienced
 More than 5 Years → Senior Employee
-----------------------------------------------------------------------------------------------

SET SERVEROUTPUT ON;
ALTER TABLE EMP
ADD JOINING_DATE DATE;
UPDATE EMP
SET JOINING_DATE = DATE '2025-01-01'
WHERE EID = 101;

UPDATE EMP
SET JOINING_DATE = DATE '2023-01-01'
WHERE EID = 102;

UPDATE EMP
SET JOINING_DATE = DATE '2018-01-01'
WHERE EID = 103;

COMMIT;

CREATE OR REPLACE FUNCTION EMP_EXPERIENCE(
P_EMP_ID IN EMP.EID%TYPE
)
RETURN VARCHAR2
IS
V_JOINING_DATE EMP.JOINING_DATE%TYPE;
V_YEARS NUMBER;
BEGIN
SELECT JOINING_DATE
INTO V_JOINING_DATE
FROM EMP
WHERE EID = P_EMP_ID;
V_YEARS := MONTHS_BETWEEN(SYSDATE, V_JOINING_DATE) / 12;
IF V_YEARS < 2 THEN
RETURN 'FRESHER';
ELSIF V_YEARS <= 5 THEN

RETURN 'EXPERIENCED';
ELSE
RETURN 'SENIOR EMPLOYEE';
END IF;
EXCEPTION

WHEN NO_DATA_FOUND THEN
RETURN 'EMPLOYEE ID NOT FOUND';
END;
/
SELECT EID,
ENAME,
JOINING_DATE,
EMP_EXPERIENCE(EID) AS EXPERIENCE_CATEGORY
FROM EMP;
