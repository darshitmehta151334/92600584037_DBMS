8. Create a procedure to update the salary of an employee using the
employee ID. Handle the exception if the employee ID does not exist.
------------------------------------------------------------------------------------------

CREATE TABLE EMPLOYEE (
EMP_ID NUMBER,
EMP_NAME VARCHAR2(50),
SALARY NUMBER
);
INSERT INTO EMPLOYEE VALUES (101, 'DARSHIT', 25000);
INSERT INTO EMPLOYEE VALUES (102, 'HIT', 30000);

COMMIT;
CREATE OR REPLACE PROCEDURE UPDATE_SALARY(
P_EMP_ID IN NUMBER,
P_SALARY IN NUMBER
)
IS
BEGIN
UPDATE EMPLOYEE
SET SALARY = P_SALARY
WHERE EMP_ID = P_EMP_ID;
IF SQL%ROWCOUNT = 0 THEN
RAISE NO_DATA_FOUND;
END IF;

DBMS_OUTPUT.PUT_LINE('SALARY UPDATED SUCCESSFULLY.');
EXCEPTION
WHEN NO_DATA_FOUND THEN
DBMS_OUTPUT.PUT_LINE('EMPLOYEE ID DOES NOT EXIST.');
END;
/
-- Enable Output
SET SERVEROUTPUT ON;
-- Execute Procedure
BEGIN
UPDATE_SALARY(101, 50000);
END;
/
SELECT * FROM EMPLOYEE;
