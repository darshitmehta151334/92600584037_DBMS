wap to calculate grade

SET SERVEROUTPUT ON;

DECLARE
    s1 NUMBER;
    s2 NUMBER;
    s3 NUMBER;
    total NUMBER;
    per NUMBER;

BEGIN
    s1 := &s1;
    s2 := &s2;
    s3 := &s3;

    total := s1 + s2 + s3;
    per := total / 3;

    DBMS_OUTPUT.PUT_LINE('Total:- ' || total);
    DBMS_OUTPUT.PUT_LINE('Percentage:- ' || per);

   IF per >= 90 THEN
       DBMS_OUTPUT.PUT_LINE('A Grade');
   ELSIF per >= 75 THEN
      DBMS_OUTPUT.PUT_LINE('B Grade');
   ELSIF per >= 60 THEN
      DBMS_OUTPUT.PUT_LINE('C Grade');
   ELSIF per >= 35 THEN
      DBMS_OUTPUT.PUT_LINE('Pass');
   ELSE
      DBMS_OUTPUT.PUT_LINE('Fail');
   END IF;	
END;
/
